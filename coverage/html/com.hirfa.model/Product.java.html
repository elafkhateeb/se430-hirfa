package com.hirfa.model;

public class Product {

    public int stock;

    public Product(int id, int artisanId, String name, String craft, double price, int stock) {
        this.id = id;
        this.artisanId = artisanId;
        this.name = name;
        this.craft = craft;
        this.price = price;
        this.stock = stock;
    }
